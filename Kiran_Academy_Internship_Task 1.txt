Part A - Database & Table Creation

CREATE DATABASE company_db;
USE company_db;
CREATE TABLE employees (employee_id INT PRIMARY KEY, employee_name VARCHAR(100), department VARCHAR(50), salary DECIMAL(10,2), city VARCHAR(50), joining_date DATE, status VARCHAR(20));
DESC employees;

Part G - Table-Level DDL Practice

CREATE TABLE departments (department_id INT PRIMARY KEY, department_name VARCHAR(100), location VARCHAR(100));
